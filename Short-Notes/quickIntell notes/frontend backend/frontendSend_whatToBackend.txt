**How Backend Determines Permissions:**

```16:65:src/rcm/auth/ability.ts
export async function buildAbilityFor(userId: string, organizationId: string, context: any): Promise<AppAbility> {
  const { can, build } = new AbilityBuilder<AppAbility>(createPrismaAbility as any)

  const membership = await context.entities.OrganizationMember.findFirst({
    where: { userId, organizationId }
  })

  if (!membership) {
    // No access to anything if not a member
    return build()
  }

  // Owners: full implicit permissions across tasks within the organization
  if (membership.role === 'owner') {
    can('manage', 'Eob2EraRecord', { organizationId })
    can('manage', 'OrganizationMember', { organizationId })
    can('update', 'Organization', { id: organizationId })
    can('read', 'Organization', { id: organizationId })
  }

  // Note: Non-owner roles carry no implicit permissions; all access below is controlled by fine-grained permissions.

  // Fine-grained permissions
  if (membership.recordPermission === PermissionLevel.READ || membership.recordPermission === PermissionLevel.WRITE) {
    can('read', 'Eob2EraRecord', { organizationId })
  }
  if (membership.recordPermission === PermissionLevel.WRITE) {
    can('update', 'Eob2EraRecord', { organizationId })
    can('delete', 'Eob2EraRecord', { organizationId })
  }
  if (membership.uploadRecordPermission === PermissionLevel.WRITE) {
    can('create', 'Eob2EraRecord', { organizationId })
  }
  if (membership.manageUserPermission === PermissionLevel.READ || membership.manageUserPermission === PermissionLevel.WRITE) {
    can('read', 'OrganizationMember', { organizationId })
  }
  if (membership.manageUserPermission === PermissionLevel.WRITE) {
    can('create', 'OrganizationMember', { organizationId })
    can('update', 'OrganizationMember', { organizationId })
    can('delete', 'OrganizationMember', { organizationId })
  }
  if (membership.organizationDetailsPermission === PermissionLevel.READ || membership.organizationDetailsPermission === PermissionLevel.WRITE) {
    can('read', 'Organization', { id: organizationId })
  }
  if (membership.organizationDetailsPermission === PermissionLevel.WRITE) {
    can('update', 'Organization', { id: organizationId })
  }

  return build()
}
```

```160:203:src/rcm/organization/operations.ts
export const inviteMember: InviteMember<{ organizationId: string; email: string; role: string; permissions?: { record: PermissionLevel; uploadRecord: PermissionLevel; manageUser: PermissionLevel; organizationDetails: PermissionLevel } }, OrganizationMember> = async (args, context) => {
  if (!context.user) {
    throw new HttpError(401, 'Not authorized');
  }

  // Enforce via CASL ability: must be allowed to create OrganizationMember
  const ability = await buildAbilityFor(context.user.id, args.organizationId, context)
  if (!ability.can('create', 'OrganizationMember')) {
    throw new HttpError(403, 'Insufficient permissions');
  }

  // Find user by email
  const user = await context.entities.User.findFirst({
    where: { email: args.email }
  });

  if (!user) {
    throw new HttpError(404, 'User not found');
  }

  // Check if user is already a member
  const existingMembership = await context.entities.OrganizationMember.findFirst({
    where: {
      userId: user.id,
      organizationId: args.organizationId
    }
  });

  if (existingMembership) {
    throw new HttpError(400, 'User is already a member of this organization');
  }

  return context.entities.OrganizationMember.create({
    data: {
      userId: user.id,
      organizationId: args.organizationId,
      role: args.role as any,
      recordPermission: args.permissions?.record ?? PermissionLevel.NONE,
      uploadRecordPermission: args.permissions?.uploadRecord ?? PermissionLevel.NONE,
      manageUserPermission: args.permissions?.manageUser ?? PermissionLevel.NONE,
      organizationDetailsPermission: args.permissions?.organizationDetails ?? PermissionLevel.NONE
    }
  });
}
```

```81:160:src/rcm/eob2era/operations.ts
export const getEobRecords: GetEobRecords<GetEobRecordsArgs, { records: Eob2EraRecord[]; total: number; page: number; pageSize: number }> = async (args, context) => {
  if (!context.user) {
    throw new HttpError(401, 'You must be logged in to view records');
  }

  // Get user's organization
  const organization = await getUserOrganization(context.user.id, context);

  // Ability check: must be allowed to read records
  const ability = await buildAbilityFor(context.user.id, organization.id, context)
  if (!ability.can('read', 'Eob2EraRecord')) {
    throw new HttpError(403, 'You do not have permission to view records');
  }

  // Build where clause
  const where: any = {
    organizationId: organization.id,
    ...accessibleBy(ability).Eob2EraRecord,
  };

  if (args.status) {
    where.status = args.status;
  }

  if (args.payer) {
    where.payer = { contains: args.payer, mode: 'insensitive' };
  }

  if (args.facility) {
    where.facility = { contains: args.facility, mode: 'insensitive' };
  }

  if (args.searchTerm) {
    where.OR = [
      { eob: { contains: args.searchTerm, mode: 'insensitive' } },
      { payer: { contains: args.searchTerm, mode: 'insensitive' } },
      { facility: { contains: args.searchTerm, mode: 'insensitive' } },
      { era: { contains: args.searchTerm, mode: 'insensitive' } },
    ];
  }

  if (args.startDate || args.endDate) {
    where.received = {};
    if (args.startDate) {
      where.received.gte = new Date(args.startDate);
    }
    if (args.endDate) {
      where.received.lte = new Date(args.endDate);
    }
  }

  // Pagination
  const page = args.page || 1;
  const pageSize = args.pageSize || 20;
  const skip = (page - 1) * pageSize;

  // Sorting
  const orderBy: any = {};
  const sortBy = args.sortBy || 'createdAt';
  const sortOrder = args.sortOrder || 'desc';
  orderBy[sortBy] = sortOrder;

  // Fetch records
  const [records, total] = await Promise.all([
    context.entities.Eob2EraRecord.findMany({
      where,
      orderBy,
      skip,
      take: pageSize,
    }),
    context.entities.Eob2EraRecord.count({ where }),
  ]);

  return {
    records,
    total,
    page,
    pageSize,
  };
};
```

## Integration with Other Work Domains (Medical Coding, Payments)

For other domains, follow the same pattern:

1. **Add permission fields to `OrganizationMember`** for the new domain:
   ```prisma
   model OrganizationMember {
     // ... existing fields ...
     medicalCodingPermission   PermissionLevel @default(NONE)
     paymentsPermission        PermissionLevel @default(NONE)
   }
   ```

2. **Update `ability.ts`** to include the new permissions:
   ```typescript
   if (membership.medicalCodingPermission === PermissionLevel.READ || ...) {
     can('read', 'MedicalCode', { organizationId })
   }
   if (membership.paymentsPermission === PermissionLevel.WRITE) {
     can('create', 'Payment', { organizationId })
   }
   ```

3. **Create operations** for the new domain in a new operations file

4. **Use the same authorization pattern** in those operations

This approach ensures consistent, secure authorization across all domains in your application!
